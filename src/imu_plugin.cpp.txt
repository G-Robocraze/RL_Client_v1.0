#include <Arduino.h>
#include <Wire.h>
#include <MPU9250_asukiaaa.h>

// === MPU9250 Configuration ===
constexpr uint8_t MPU_ADDRESS = 0x68;  // Set to 0x69 if AD0 is high
MPU9250_asukiaaa imuSensor(MPU_ADDRESS);

// === Tilt, Vibration, and Shock Detection Thresholds ===
constexpr float MAX_TILT_DEGREES = 90.0;
constexpr float MAX_VIBRATION_RAW = 1.0;      // Max delta expected per frame
constexpr float SHOCK_ACCEL_THRESHOLD = 2.5;  // g-force threshold for crash

// === Previous Frame Values for Vibration Calculation ===
float prevAccelX = 0, prevAccelY = 0, prevAccelZ = 0;

// === Utility: Compute tilt percentage ===
float computeTiltPercentage(float roll, float pitch) {
  float tiltAngle = sqrt(sq(roll) + sq(pitch));
  tiltAngle = constrain(tiltAngle, 0, MAX_TILT_DEGREES);
  return (tiltAngle / MAX_TILT_DEGREES) * 100.0;
}

// === Utility: Compute vibration intensity (0–10) ===
float computeVibration(float ax, float ay, float az) {
  float vibrationMagnitude = sqrt(
    sq(ax - prevAccelX) +
    sq(ay - prevAccelY) +
    sq(az - prevAccelZ)
  );

  // Update previous values
  prevAccelX = ax;
  prevAccelY = ay;
  prevAccelZ = az;

  float vibrationNormalized = constrain(vibrationMagnitude / MAX_VIBRATION_RAW, 0.0, 1.0);
  return vibrationNormalized * 10.0;  // Scale to 0–10
}

// === Utility: Detect crash/fall/shock ===
bool detectCrash(float ax, float ay, float az) {
  float totalAccel = sqrt(ax * ax + ay * ay + az * az);
  return totalAccel > SHOCK_ACCEL_THRESHOLD;
}

// === Setup ===
void setup() {
  Serial.begin(115200);
  while (!Serial);

  Wire.begin();  // MKR WAN 1310 uses SDA: D11, SCL: D12
  imuSensor.setWire(&Wire);

  imuSensor.beginAccel();
  imuSensor.beginGyro();
  imuSensor.beginMag();  // Optional: not used here

  Serial.println("MPU9250 Initialized");
}

// === Main Loop ===
void loop() {
  // Update sensor values
  imuSensor.accelUpdate();
  imuSensor.gyroUpdate();

  float ax = imuSensor.accelX();
  float ay = imuSensor.accelY();
  float az = imuSensor.accelZ();

  // === Tilt Calculation (Roll, Pitch, Tilt %)
  float roll  = atan2(ay, az) * 180 / PI;
  float pitch = atan2(-ax, sqrt(ay * ay + az * az)) * 180 / PI;
  float tiltPercentage = computeTiltPercentage(roll, pitch);

  // === Vibration Calculation
  float vibration = computeVibration(ax, ay, az);

  // === Crash/Fall Detection
  bool crashDetected = detectCrash(ax, ay, az);

  // === Output ===
  Serial.println("---- Sensor Data ----");
  Serial.print("Accel (g): ");
  Serial.print(ax, 3); Serial.print(", ");
  Serial.print(ay, 3); Serial.print(", ");
  Serial.println(az, 3);

  Serial.print("Tilt (%): ");
  Serial.println(tiltPercentage, 2);

  Serial.print("Vibration Index (0–10): ");
  Serial.println(vibration, 2);

  if (crashDetected) {
    Serial.println("Crash/Fall Detected!");
  }

  Serial.println();
  delay(100);  // 10 Hz update rate
}
